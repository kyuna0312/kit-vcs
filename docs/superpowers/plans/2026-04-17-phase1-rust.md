```rust
    ObjectNotFound(String),

}

impl std::fmt::Display for KitError {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        match self {
            KitError::Io(e)             => write!(f, "IO error: {}", e),
            KitError::NotARepo          => write!(f, "not a kit repository"),
            KitError::ObjectNotFound(h) => write!(f, "object not found: {}", h),
            KitError::InvalidFormat(s)  => write!(f, "invalid format: {}", s),
            KitError::Message(s)        => write!(f, "{}", s),
        }
    }
}

impl From<std::io::Error> for KitError {
    fn from(e: std::io::Error) -> Self { KitError::Io(e) }
}

pub type Result<T> = std::result::Result<T, KitError>;
```

```rust
use crate::utils::hash::sha1_hex;
use crate::core::{KitError, Result};

pub struct Blob {
    pub content: Vec<u8>,
}

impl Blob {
    pub fn serialize(&self) -> Vec<u8> {
        let header = format!("blob {}\n", self.content.len());
        let mut out = header.into_bytes();
        out.extend_from_slice(&self.content);
        out
    }

    pub fn hash(&self) -> String {
        sha1_hex(&self.serialize())
    }

    pub fn deserialize(raw: &[u8]) -> Result<Self> {
        let nl = raw.iter().position(|&b| b == b'\n')
            .ok_or_else(|| KitError::InvalidFormat("missing newline in blob".into()))?;
        let header = std::str::from_utf8(&raw[..nl])
            .map_err(|_| KitError::InvalidFormat("blob header not utf8".into()))?;
        if !header.starts_with("blob ") {
            return Err(KitError::InvalidFormat(format!("expected blob header, got: {}", header)));
        }
        Ok(Blob { content: raw[nl + 1..].to_vec() })
    }

    pub fn from_file(path: &std::path::Path) -> Result<Self> {
        let content = crate::utils::fs_utils::read_file(path)?;
        Ok(Blob { content })
    }
}
```

```rust
use crate::utils::hash::sha1_hex;
use crate::core::{KitError, Result};

pub struct TreeEntry {
    pub mode: String,
    pub name: String,
    pub hash: String,
}

pub struct Tree {
    pub entries: Vec<TreeEntry>,
}

impl Tree {
    pub fn sort(&mut self) {
        self.entries.sort_by(|a, b| a.name.cmp(&b.name));
    }

    pub fn serialize(&self) -> Vec<u8> {
        let mut body = Vec::new();
        for e in &self.entries {
            body.extend_from_slice(format!("{} {}\0{}", e.mode, e.name, e.hash).as_bytes());
            body.push(b'\n');
        }
        let header = format!("tree {}\n", self.entries.len());
        let mut out = header.into_bytes();
        out.extend_from_slice(&body);
        out
    }

    pub fn hash(&self) -> String {
        sha1_hex(&self.serialize())
    }

    pub fn deserialize(raw: &[u8]) -> Result<Self> {
        let text = std::str::from_utf8(raw)
            .map_err(|_| KitError::InvalidFormat("tree not utf8".into()))?;
        let mut lines = text.lines();
        let header = lines.next()
            .ok_or_else(|| KitError::InvalidFormat("empty tree".into()))?;
        if !header.starts_with("tree ") {
            return Err(KitError::InvalidFormat(format!("expected tree header: {}", header)));
        }
        let mut entries = Vec::new();
        for line in lines {
            if line.is_empty() { continue; }
            // format: "mode name\0hash"
            let null_pos = line.find('\0')
                .ok_or_else(|| KitError::InvalidFormat("tree entry missing null".into()))?;
            let mode_name = &line[..null_pos];
            let hash = &line[null_pos + 1..];
            let space = mode_name.find(' ')
                .ok_or_else(|| KitError::InvalidFormat("tree entry missing space".into()))?;
            entries.push(TreeEntry {
                mode: mode_name[..space].to_string(),
                name: mode_name[space + 1..].to_string(),
                hash: hash.to_string(),
            });
        }
        Ok(Tree { entries })
    }
}
```

```rust
use crate::utils::hash::sha1_hex;
use crate::core::{KitError, Result};

pub struct Commit {
    pub tree:      String,
    pub parent:    Option<String>,
    pub author:    String,
    pub timestamp: i64,
    pub message:   String,
}

impl Commit {
    pub fn serialize(&self) -> Vec<u8> {
        let mut body = format!("tree {}\n", self.tree);
        if let Some(p) = &self.parent {
            body.push_str(&format!("parent {}\n", p));
        }
        body.push_str(&format!("author {}\n", self.author));
        body.push_str(&format!("timestamp {}\n", self.timestamp));
        body.push('\n');
        body.push_str(&self.message);
        body.push('\n');

        let header = format!("commit {}\n", body.len());
        let mut out = header.into_bytes();
        out.extend_from_slice(body.as_bytes());
        out
    }

    pub fn hash(&self) -> String {
        sha1_hex(&self.serialize())
    }

    pub fn deserialize(raw: &[u8]) -> Result<Self> {
        let text = std::str::from_utf8(raw)
            .map_err(|_| KitError::InvalidFormat("commit not utf8".into()))?;
        let mut lines = text.lines();

        // skip "commit <size>" header line
        let header = lines.next()
            .ok_or_else(|| KitError::InvalidFormat("empty commit".into()))?;
        if !header.starts_with("commit ") {
            return Err(KitError::InvalidFormat(format!("expected commit header: {}", header)));
        }

        let mut tree = String::new();
        let mut parent = None;
        let mut author = String::new();
        let mut timestamp = 0i64;
        let mut in_message = false;
        let mut message_lines = Vec::new();

        for line in lines {
            if in_message {
                message_lines.push(line);
                continue;
            }
            if line.is_empty() { in_message = true; continue; }
            if let Some(v) = line.strip_prefix("tree ")      { tree = v.to_string(); }
            else if let Some(v) = line.strip_prefix("parent ")   { parent = Some(v.to_string()); }
            else if let Some(v) = line.strip_prefix("author ")   { author = v.to_string(); }
            else if let Some(v) = line.strip_prefix("timestamp ") {
                timestamp = v.parse().map_err(|_| KitError::InvalidFormat("bad timestamp".into()))?;
            }
        }

        Ok(Commit { tree, parent, author, timestamp, message: message_lines.join("\n").trim_end_matches('\n').to_string() })
    }
}
```

```rust
use std::path::PathBuf;
use crate::core::{KitError, Result};
use crate::utils::fs_utils::{read_string, write_file};

pub struct Refs {
    kit_dir: PathBuf,
}

impl Refs {
    pub fn new(kit_dir: PathBuf) -> Self { Refs { kit_dir } }

    pub fn head_path(&self) -> PathBuf { self.kit_dir.join("HEAD") }
    pub fn branch_path(&self, branch: &str) -> PathBuf {
        self.kit_dir.join("refs").join("heads").join(branch)
    }

    pub fn read_head(&self) -> Result<String> {
        Ok(read_string(&self.head_path())?.trim().to_string())
    }

    pub fn write_head(&self, content: &str) -> Result<()> {
        write_file(&self.head_path(), format!("{}\n", content).as_bytes())?;
        Ok(())
    }

    pub fn current_branch(&self) -> Result<Option<String>> {
        let head = self.read_head()?;
        if let Some(branch) = head.strip_prefix("ref: refs/heads/") {
            Ok(Some(branch.to_string()))
        } else {
            Ok(None) // detached HEAD
        }
    }

    pub fn read_branch(&self, branch: &str) -> Result<Option<String>> {
        let path = self.branch_path(branch);
        if !path.exists() { return Ok(None); }
        Ok(Some(read_string(&path)?.trim().to_string()))
    }

    pub fn write_branch(&self, branch: &str, hash: &str) -> Result<()> {
        let path = self.branch_path(branch);
        write_file(&path, format!("{}\n", hash).as_bytes())?;
        Ok(())
    }

    pub fn head_commit(&self) -> Result<Option<String>> {
        let head = self.read_head()?;
        if let Some(branch) = head.strip_prefix("ref: refs/heads/") {
            self.read_branch(branch)
        } else if head.len() == 40 {
            Ok(Some(head))
        } else {
            Ok(None)
        }
    }

    pub fn list_branches(&self) -> Result<Vec<String>> {
        let heads_dir = self.kit_dir.join("refs").join("heads");
        if !heads_dir.exists() { return Ok(vec![]); }
        let mut branches = Vec::new();
        for entry in std::fs::read_dir(&heads_dir)? {
            let entry = entry?;
            branches.push(entry.file_name().to_string_lossy().to_string());
        }
        branches.sort();
        Ok(branches)
    }
}
```

```rust
// rust/src/core/index.rs
use std::path::PathBuf;
use crate::core::Result;
use crate::utils::fs_utils::{read_string, write_file};

pub struct Index {
    path: PathBuf,
}

impl Index {
    pub fn new(path: PathBuf) -> Self { Index { path } }

    pub fn read(&self) -> Result<Vec<String>> {
        if !self.path.exists() { return Ok(vec![]); }
        let content = read_string(&self.path)?;
        Ok(content.lines().filter(|l| !l.is_empty()).map(|l| l.to_string()).collect())
    }

    pub fn write(&self, entries: &[String]) -> Result<()> {
        let content = entries.join("\n") + if entries.is_empty() { "" } else { "\n" };
        write_file(&self.path, content.as_bytes())?;
        Ok(())
    }

    pub fn add(&self, path: &str) -> Result<()> {
        let mut entries = self.read()?;
        if !entries.contains(&path.to_string()) {
            entries.push(path.to_string());
            entries.sort();
        }
        self.write(&entries)
    }
}
```

```rust
// rust/src/core/repository.rs
use std::path::{Path, PathBuf};
use crate::core::{KitError, Result};
use crate::core::refs::Refs;
use crate::core::index::Index;
use crate::utils::fs_utils::write_file;

pub struct Repository {
    path: PathBuf,
    pub refs: Refs,
}

impl Repository {
    pub fn init(path: &Path) -> Result<()> {
        let kit = path.join(".kit");
        std::fs::create_dir_all(kit.join("objects"))?;
        std::fs::create_dir_all(kit.join("refs").join("heads"))?;
        write_file(&kit.join("HEAD"), b"ref: refs/heads/master\n")?;
        write_file(&kit.join("index"), b"")?;
        Ok(())
    }

    pub fn open(path: &Path) -> Result<Self> {
        let kit = path.join(".kit");
        if !kit.exists() { return Err(KitError::NotARepo); }
        let refs = Refs::new(kit);
        Ok(Repository { path: path.to_path_buf(), refs })
    }

    pub fn path(&self) -> &Path { &self.path }
    pub fn kit_dir(&self) -> PathBuf { self.path.join(".kit") }
    pub fn objects_dir(&self) -> PathBuf { self.kit_dir().join("objects") }
    pub fn index(&self) -> Index { Index::new(self.kit_dir().join("index")) }

    pub fn write_object(&self, hash: &str, data: &[u8]) -> Result<()> {
        let path = self.objects_dir().join(hash);
        write_file(&path, data)?;
        Ok(())
    }

    pub fn read_object(&self, hash: &str) -> Result<Vec<u8>> {
        let path = self.objects_dir().join(hash);
        if !path.exists() { return Err(KitError::ObjectNotFound(hash.to_string())); }
        Ok(std::fs::read(&path)?)
    }

    pub fn object_exists(&self, hash: &str) -> bool {
        self.objects_dir().join(hash).exists()
    }
}
```

```rust
// rust/src/cli/commands/cmd_commit.rs
use crate::core::{Result, KitError};
use crate::core::repository::Repository;
use crate::core::objects::{blob::Blob, tree::{Tree, TreeEntry}, commit::Commit};
use std::path::Path;

pub fn run(message: String, author: String) -> Result<()> {
    let cwd = std::env::current_dir()?;
    let repo = Repository::open(&cwd)?;
    let staged = repo.index().read()?;
    if staged.is_empty() {
        return Err(KitError::Message("nothing to commit".into()));
    }

    // build tree
    let mut entries = Vec::new();
    for file in &staged {
        let blob = Blob::from_file(Path::new(file))?;
        let hash = blob.hash();
        repo.write_object(&hash, &blob.serialize())?;
        entries.push(TreeEntry { mode: "100644".into(), name: file.clone(), hash });
    }
    let mut tree = crate::core::objects::tree::Tree { entries };
    tree.sort();
    let tree_hash = tree.hash();
    repo.write_object(&tree_hash, &tree.serialize())?;

    // build commit
    let parent = repo.refs.head_commit()?;
    let timestamp = std::time::SystemTime::now()
        .duration_since(std::time::UNIX_EPOCH).unwrap().as_secs() as i64;
    let commit = Commit { tree: tree_hash, parent, author, timestamp, message: message.clone() };
    let commit_hash = commit.hash();
    repo.write_object(&commit_hash, &commit.serialize())?;

    // update HEAD ref
    let branch = repo.refs.current_branch()?.unwrap_or("master".into());
    repo.refs.write_branch(&branch, &commit_hash)?;

    println!("[{}] {}", &commit_hash[..7], message);
    Ok(())
}
```
